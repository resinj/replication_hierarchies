Example 2.8
k = 3
n = 4

f = generate_forecasts(k,n)

sle_DC = construct_sle_DC(k,n,f)

g_hom = solve_hom(sle_DC)
g_part = sle_DC$f

c = c(0.4,0.4,0.4)
g = matrix(g_part + g_hom %*% c,nrow = k,byrow = TRUE)

fractions(t(g/n))
##      [,1] [,2] [,3]
## [1,]  3/5 7/20 1/20
## [2,] 7/20  2/5  1/4
## [3,] 3/20 3/20 7/10
test_all(k,n,f,g)
##        CwC    MC   fCC   pCC   fPC   pPC   aPC   fTC   pTC
## [1,] FALSE FALSE FALSE FALSE FALSE FALSE FALSE FALSE FALSE
plot_QC(k,n,f,g)

The figure plots upper and lower coverage (top) and predicted quantiles against quantile levels to facilitate visual verification of the quantile calibration condition.

Example 3.3 (w = 0.5)
# Construct example with 3 classes (w = 0)
k = 3
n = 4

f = generate_forecasts(k,n)

sle_DC = construct_sle_DC(k,n,f)

g_part = sle_DC$f
g_hom = solve_hom(sle_DC)

c = -c(2,0,0)
g = matrix(g_part + g_hom %*% c,nrow = k,byrow = TRUE)

# Add fourth class with probability w = 0.5 = 4/8 and rescale other probabilities
# by doubling n
k = 4
n = 8

f = rbind(f, c(4,4,4))
g = rbind(g, c(4,4,4))

sle_DC = construct_sle_DC(k,n,f)
DC = test_sle(sle_DC,g)

cbind(test_all(k,n,f,g),DC)
##        CwC    MC   fCC   pCC   fPC   pPC   aPC   fTC   pTC   DC
## [1,] FALSE FALSE FALSE FALSE FALSE FALSE FALSE FALSE FALSE TRUE
